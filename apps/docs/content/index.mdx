```ts
import { contract } from '@ts-contract/core';
import { z } from 'zod';

const api = contract({
  getUser: {
    method: 'GET',
    path: '/users/:id',
    params: z.object({ id: z.string() }),
    response: z.object({
      id: z.string(),
      name: z.string(),
      email: z.string().email(),
    }),
  },
});
```
